function JF12_variant(id) {
  for (var i = 0; i < JF12_VARIANTS.length; i++)
    if (JF12_VARIANTS[i].id === id) return JF12_VARIANTS[i];
  return null;
}
function JF12_gender() {
  var items = (typeof state !== 'undefined' && Array.isArray(state.placeholders)) ? state.placeholders : [];
  for (var i = 0; i < items.length; i++) {
    var item = items[i] || {};
    var q = String(item.question || '').toLowerCase();
    var a = String(item.answer || '').toLowerCase().trim();
    if (/gender/.test(q) && (a === 'male' || a === 'female')) return a;
  }
  return 'unknown';
}
function JF12_log(message) {
  if (JF12_CONFIG.debug && typeof log === 'function') log('[JF12] ' + message);
}
function JF12_select() {
  if (!state || typeof state !== 'object') return null;
  var saved = state.jf11;
  // Preserve previously selected variants even in pre-hotfix adventures.
  if (saved && JF12_variant(Number(saved.variantId))) {
    saved.version = '1.2.0';
    if (saved.disabled) delete saved.disabled;
    return JF12_variant(Number(saved.variantId));
  }
  var count = (typeof info !== 'undefined' && info && typeof info.actionCount === 'number') ? info.actionCount : 0;
  // v1.1 permanently disabled the script after 2 actions. Do not copy that
  // behavior: allow late first activation in the opening, but avoid injecting
  // a brand-new secret into mature pre-existing adventures.
  if (JF12_CONFIG.protectEstablished && count > JF12_CONFIG.establishedAfter) {
    JF12_log('no previous choice and actionCount=' + count + ': keeping mature adventure untouched');
    return null;
  }
  var gender = JF12_gender();
  var eligible = JF12_VARIANTS.filter(function (v) { return !v.maleOnly || gender === 'male'; });
  var forced = JF12_variant(Number(JF12_CONFIG.forceVariant));
  var chosen = (forced && (!forced.maleOnly || gender === 'male')) ? forced :
    eligible[Math.floor(Math.random() * eligible.length)];
  if (!chosen) return null;
  state.jf11 = { version: '1.2.0', variantId: chosen.id, genderAtSelection: gender };
  JF12_log('selected ' + chosen.id + ': ' + chosen.title);
  return chosen;
}
function JF12_cue(v) {
  return '[JF12_BEGIN]\n' + JF12_COMPACT[v.id - 1] + '\n[JF12_END]';
}
function JF12_memory(v) {
  if (!v) return;
  state.memory = state.memory || {};
  var before = typeof state.memory.frontMemory === 'string' ? state.memory.frontMemory : '';
  // Drop old v1.1 and current v1.2 blocks without deleting other scripts' cues.
  var clean = before.replace(/\[JF11_BEGIN\][\s\S]*?\[JF11_END\]/g, '')
                    .replace(/\[JF12_BEGIN\][\s\S]*?\[JF12_END\]/g, '');
  // An incomplete old block can arise if a memory field was truncated.
  var orphan = clean.search(/\[(?:JF11|JF12)_BEGIN\]/);
  if (orphan >= 0) clean = clean.slice(0, orphan);
  clean = clean.trim();
  var cue = JF12_cue(v);
  var result = clean ? clean + '\n\n' + cue : cue;
  if (result !== before) state.memory.frontMemory = result;
  if (result.length > 500) JF12_log('combined frontMemory is ' + result.length + ' characters; other script may be crowding it');
}
function JF12_prepare() {
  var v = JF12_select();
  if (v) JF12_memory(v);
  return v;
}
function JF12_contextFallback(text, v) {
  // Front Memory changes in Context may only affect the following turn.
  // On normal (non-optimized) models, a conservative tail fallback gives the
  // just-selected cue to the current generation when Input was bypassed.
  // Optimized-context models may discard modified Context text. Our persistent
  // frontMemory is still primed for the NEXT model call.
  if (!v || !JF12_CONFIG.contextTailFallback || typeof text !== 'string') return text;
  var marker = '[JF12_BEGIN]';
  if (text.indexOf(marker) !== -1) return text;
  var extra = '\n' + JF12_cue(v);
  var max = (typeof info !== 'undefined' && typeof info.maxChars === 'number') ? info.maxChars : 0;
  // Do not risk truncating existing context. If no reliable limit is supplied,
  // preserve the original context unchanged; frontMemory supplies future turns.
  if (!max || text.length + extra.length + 250 >= max) {
    JF12_log('context lacks marker; tail fallback skipped for capacity/unknown maxChars');
    return text;
  }
  JF12_log('context lacked cue: appended tail fallback (model may ignore on optimized context)');
  return text + extra;
}
