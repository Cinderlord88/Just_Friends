function JF11_gender() {
  var items = (typeof state !== 'undefined' && Array.isArray(state.placeholders)) ? state.placeholders : [];
  for (var i = 0; i < items.length; i++) {
    var item = items[i] || {};
    var q = String(item.question || '').toLowerCase().trim();
    if (q === 'character.gender' || /\bgender\b/.test(q)) {
      var a = String(item.answer || '').toLowerCase().trim();
      if (a === 'male' || a === 'female') return a;
    }
  }
  return 'unknown';
}

function JF11_getVariant(id) {
  for (var i = 0; i < JF11_VARIANTS.length; i++) {
    if (JF11_VARIANTS[i].id === id) return JF11_VARIANTS[i];
  }
  return null;
}

function JF11_eligible(v, gender) {
  return !v.maleOnly || gender === 'male';
}

function JF11_select() {
  if (state.jf11 && state.jf11.disabled) return null;
  if (state.jf11 && JF11_getVariant(state.jf11.variantId)) {
    return JF11_getVariant(state.jf11.variantId);
  }

  var actionCount = (typeof info !== 'undefined' && typeof info.actionCount === 'number') ? info.actionCount : 0;
  if (JF11_CONFIG.protectOldAdventures && actionCount > 2) {
    state.jf11 = { version: '1.1.0', disabled: true, reason: 'existing adventure' };
    return null;
  }

  var gender = JF11_gender();
  var eligible = JF11_VARIANTS.filter(function (v) { return JF11_eligible(v, gender); });
  var forced = JF11_getVariant(Number(JF11_CONFIG.forceVariant));
  var chosen = (forced && JF11_eligible(forced, gender)) ? forced :
    eligible[Math.floor(Math.random() * eligible.length)];

  state.jf11 = { version: '1.1.0', variantId: chosen.id, genderAtSelection: gender };
  if (JF11_CONFIG.debug && typeof log === 'function') {
    log('[JF11] variant ' + chosen.id + ': ' + chosen.title);
  }
  return chosen;
}

function JF11_cue(v) {
  return [
    '[JUST FRIENDS — PRIVATE STORY FACTS v1.1; for narrator, not player]',
    'ONE randomly chosen explanation applies for this entire adventure. Never combine variants or silently substitute a different secret.',
    'Nina Williams and the adult player have been genuinely dating exclusively for four months. She cares for the player and is NOT ashamed of them, using them, or pursuing another partner. Her words "We\'re just friends" were an impulsive mistake, not a breakup.',
    'HIDDEN FACT: ' + v.fact,
    'WHY SHE PANICKED: ' + v.reason,
    'SECOND ACT: ' + v.next,
    'PACING: Lydia\'s birthday party is ENDING in the opening. Nina can speak privately IMMEDIATELY. When the player asks why she said it, asks for the truth, or agrees to talk, Nina must give the genuine selected explanation in that reply; no prolonged evasions. Do not automatically absolve her; let the unresolved practical or social issue give a few meaningful follow-up choices.',
    'AGENCY: Never dictate the player\'s feelings, goals, sexual history, consent, reply, or forgiveness. Reconciliation must remain genuinely possible, and rejection equally permissible. Only Sophie knows the relationship at the start; other friends know only what they witnessed or were told.'
  ].join('\n');
}

function JF11_applyMemory(v) {
  // frontMemory is a documented scripting slot and works without rewriting the
  // assembled context text. Replace ONLY our own block; preserve other scripts.
  var begin = '[JF11_BEGIN]';
  var end = '[JF11_END]';
  state.memory = state.memory || {};
  var old = typeof state.memory.frontMemory === 'string' ? state.memory.frontMemory : '';
  var clean = old.replace(/\[JF11_BEGIN\][\s\S]*?\[JF11_END\]/g, '').trim();
  var block = begin + '\n' + JF11_cue(v) + '\n' + end;
  state.memory.frontMemory = clean ? clean + '\n\n' + block : block;
}
